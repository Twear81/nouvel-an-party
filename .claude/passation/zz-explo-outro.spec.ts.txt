import { mkdirSync, writeFileSync } from "node:fs";
import path from "node:path";
import { expect, test } from "./helpers/fixtures";

const OUT = process.env.OUTRO_OUT ?? "/tmp/outro-v2";
mkdirSync(OUT, { recursive: true });

const NAMES = ["Anne-Sophie Dupo", "Jean-Christophe!", "Marie-Charlotte1", "Bartholomew Jr..", "Léonard de Vinci", "Xx_DarkSasuke_xX", "Marguerite-Marie", "Fulgencio Montoy"];
const AVATARS = ["renard", "panda", "lion", "licorne", "dragon", "pieuvre", "chou-fleur", "tiramisu"];
const COLORS = ["#e8344f", "#1f6fe0", "#1fa05f", "#f08a1c", "#a259ff", "#14b8a6", "#f472b6", "#eab308"];
const FRAMES = ["gold", "none", "double", "none", "dashed", "none", "none", "gold"];
const ids = NAMES.map((_, i) => `p${i + 1}`);

const strokes = [
  { c: 1, w: 1, p: [500, 150, 850, 800, 150, 800, 500, 150] },
  { c: 2, w: 1, p: [300, 400, 330, 420, 700, 420, 730, 400] },
  { c: 3, w: 1, p: [420, 560, 580, 560, 580, 700, 420, 700, 420, 560] },

function buildSave(startedAt: number) {
  const before: Record<string, number> = { p1: 1000, p2: 900, p3: 800, p4: 800, p5: 700, p6: 450, p7: 400, p8: 300 };
  const cat = (id: string, title: string, emoji: string) => ({ id, title, emoji });
  const categories = {
    "talent-incompris": { title: "Le talent incompris", emoji: "🎨" }, "duo-improbable": { title: "Duo improbable", emoji: "🧲" },
    "merci-travaux": { title: "Merci pour les travaux", emoji: "🏗️" }, escargot: { title: "L'escargot", emoji: "🐌" },
    "roi-manches": { title: "Roi des manches", emoji: "👑" }, "craquage-dos": { title: "Le craquage de dos", emoji: "🦴" },
  };
  void cat;
  const award = (id: string, phrase: string, winners: string[], statLabel: string, roulette: string[], extra = {}) =>
    ({ id, title: (categories as any)[id].title, emoji: (categories as any)[id].emoji, phrase, winners, points: 120, statLabel, roulette, ...extra });
  const drawing = { kind: "dessin", tag: "incompris", playerId: "p8", label: "Un chat qui fait du vélo", strokes, score: 0 };
  const ceremony = {
    startedAt, playerIds: ids, scoresBefore: before, timeScale: 1, pausedMs: 0, pausedSince: null,
    souvenirs: [
      { ...drawing, tag: "meilleur", playerId: "p3", label: "Une tour Eiffel un peu penchée", score: 6 },
      { kind: "objet", emoji: "🎮", label: "PlayStation 5", detail: "649,99 €", score: 210 },
      { kind: "post-it", text: "Quel animal ne dort jamais ?", detail: "Le requin — 5/8 réponses", score: 0.6 },
    ],
    awards: [
      award("talent-incompris", "Le public n'était pas prêt.", ["p8"], "0 vote sur 3 dessins", ["roi-manches", "escargot", "craquage-dos", "merci-travaux", "talent-incompris"], { keepsake: drawing }),
      award("duo-improbable", "Ont passé une heure ensemble sans partager une seule opinion.", ["p3", "p6"], "0 avis en commun", ["escargot", "roi-manches", "craquage-dos", "duo-improbable"]),
      award("merci-travaux", "Sans fondations, aucun podium.", ["p6"], "3 dernières places", ["craquage-dos", "talent-incompris", "escargot", "merci-travaux"]),
      award("escargot", "Le temps, c'est de l'argent. Pas pour lui.", ["p5"], "812 ms en moyenne", ["merci-travaux", "roi-manches", "talent-incompris", "escargot"]),
    ],
    categories,
  };
  return {
    version: 1,
    players: ids.map((id, i) => ({ id, token: `tok-${id}`, score: before[id], profile: { name: NAMES[i], avatar: { kind: "preset", id: AVATARS[i] }, frame: FRAMES[i], color: COLORS[i] } })),
    setup: { counts: {} }, plan: [{ gameId: "ceci-ou-cela", item: {}, skin: "dofus" }], roundIndex: 0, phase: "ceremony",
    history: [], ceremony, finalReveal: null,
  };
}

test.describe("explo outro", () => {
  test.use({ serverOptions: { outro: true, ceremonyTimeScale: 1, finalTimeScale: 1 } });
  test("cérémonie et classement final", async ({ party, browser, phoneOptions }) => {
    test.setTimeout(400_000);
    const { tv, server } = party;
    const shot = async (name: string, page = tv) => { await page.screenshot({ path: path.join(OUT, `${name}.png`) }); };
    await server.kill();
    writeFileSync(path.join(server.dataDir, "save.json"), JSON.stringify(buildSave(Date.now() + 14_000)));
    await server.start();
    await tv.getByTestId("resume-game").click({ timeout: 20_000 });

    // Téléphones de Marie-Charlotte (duo) et Xx_DarkSasuke (duo + merci), 360x640.
    const phones = [] as import("@playwright/test").Page[];
    for (const id of ["p3", "p6", "p7"]) {
      const context = await browser.newContext({ ...phoneOptions, viewport: { width: 360, height: 640 }, ignoreHTTPSErrors: true });
      await context.addInitScript((t) => localStorage.setItem("nouvel-an-token", t), `tok-${id}`);
      const page = await context.newPage();
      await page.goto(server.playUrl);
      phones.push(page);
    }
    // Observateur côté page : date la première apparition de chaque valeur d'attribut d'étape.
    await tv.evaluate(() => {
      const w = window as any; w.__marks = {} as Record<string, number>;
      const note = () => {
        const c = document.querySelector('[data-testid="tv-ceremony"]');
        if (c) { const k = `c:${c.getAttribute("data-step")}:${c.getAttribute("data-index") ?? ""}`; w.__marks[k] ??= performance.now(); }
        const f = document.querySelector('[data-testid="tv-final"]');
        if (f) { w.__marks["f:start"] ??= performance.now(); if (f.getAttribute("data-stage") === "done") w.__marks["f:done"] ??= performance.now(); }
        const card = document.querySelector('[data-testid="final-card"]');
        if (card) { const k = `card:${card.getAttribute("data-rank")}:${card.getAttribute("data-phase")}`; w.__marks[k] ??= performance.now(); }
        for (const id of ["podium", "podium-1", "podium-2", "podium-3", "tv-screen"]) if (document.querySelector(`[data-testid="${id}"]`)) w.__marks[`t:${id}`] ??= performance.now();
      };
      new MutationObserver(note).observe(document.body, { subtree: true, childList: true, attributes: true });
      note();
    });
    /** Capture `ms` après la première apparition de `key` (rien si c'est déjà trop tard de plus d'une seconde). */
    const when = async (key: string, ms: number, name: string, page = tv) => {
      await page.waitForFunction((k) => (window as any).__marks[k] !== undefined, key, { timeout: 90_000 });
      const wait = await page.evaluate(([k, d]) => Math.max(-9999, (d as number) - (performance.now() - (window as any).__marks[k as string])), [key, ms]);
      if (wait < -1000) { console.log("trop tard", name); return; }
      if (wait > 0) await page.waitForTimeout(wait);
      await shot(name, page);
    };

    await when("c:intro:", 2200, "01-intro");
    for (let i = 0; i < 3; i++) {
      await when(`c:souvenir:${i}`, 900, `02-souvenir${i}-a-zoom`);
      await when(`c:souvenir:${i}`, 2300, `02-souvenir${i}-b-pose`);
      await when(`c:souvenir:${i}`, 3900, `02-souvenir${i}-c-legende`);
      await when(`c:souvenir:${i}`, 5900, `02-souvenir${i}-d-fin`);
    }
    await when("c:provisoire:", 3500, "03-provisoire");
    for (let i = 0; i < 4; i++) {
      await when(`c:roulette:${i}`, 600, `04-award${i}-a-enveloppe`);
      await when(`c:roulette:${i}`, 2500, `04-award${i}-b-roulette`);
      await when(`c:carte:${i}`, 1500, `04-award${i}-c-carte`);
      await when(`c:projecteur:${i}`, 1500, `04-award${i}-d-projecteur`);
      await when(`c:gagnant:${i}`, 1100, `04-award${i}-e-gagnant`);
      if (i === 1) await shot("ph-duo", phones[0]!);
      await when(`c:gagnant:${i}`, 3400, `04-award${i}-f-etoile`);
    }
    await shot("ph-ceremonie-p6", phones[1]!);
    await shot("ph-ceremonie-p7", phones[2]!);

    // Classement final
    await when("f:start", 1500, "05-final-intro");
    await when("card:8:count", 900, "05-final-8-count");
    await when("card:7:stars", 500, "05-final-7-stars");
    await when("card:6:stars", 450, "05-final-6-stars-a");
    await when("card:6:stars", 1500, "05-final-6-stars-b");
    await when("card:5:advance", 300, "05-final-5-advance");
    await when("t:podium", 1300, "06-podium-monte");
    await when("card:3:stars", 500, "06-final-3-stars");
    await when("t:podium-3", 1500, "06-podium-3");
    await when("t:podium-2", 1500, "06-podium-2");
    await shot("ph-final-attente-p3", phones[0]!);
    await when("t:tv-screen", 1800, "07-roulement");
    await when("t:podium-1", 1500, "08-premier");
    await when("t:podium-1", 4500, "08-premier-b");
    await when("f:done", 1500, "09-final-etat-final");
    await shot("ph-final-p3", phones[0]!);
    await shot("ph-final-p7", phones[2]!);
  });
});
